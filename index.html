Angka sesuai nomor modul pada papan trainer. Cara konvensional memakai modul 3 (kunci kontak) pada posisi START.

#WaktuDurasiKeterangan
#include <WiFi.h>
#include <WebServer.h>

const int RELAY_PIN = 26;        // ke modul 4: relay optocoupler
WebServer server(80);
bool engineOn = false;

void setEngine(bool on) {
  engineOn = on;
  digitalWrite(RELAY_PIN, on ? HIGH : LOW);  // tukar HIGH/LOW jika relai aktif-LOW
  server.send(200, "text/plain", on ? "ON" : "OFF");
}

void setup() {
  pinMode(RELAY_PIN, OUTPUT);
  digitalWrite(RELAY_PIN, LOW);                 // aman: mati saat dinyalakan
  WiFi.softAP("NETRO-Trainer-01", "netro1234"); // IP trainer: 192.168.4.1
  server.on("/on",     [] { setEngine(true);  });
  server.on("/off",    [] { setEngine(false); });
  server.on("/status", [] { server.send(200, "text/plain", engineOn ? "ON" : "OFF"); });
  server.begin();
}

void loop() {
  server.handleClient();
}
